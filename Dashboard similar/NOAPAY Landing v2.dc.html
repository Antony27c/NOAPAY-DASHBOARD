import { NoaClient, FiatCurrency } from "noa-pay";

const noa = new NoaClient({
  mercadopago: { accessToken, sandbox: true },
  settlement: async ({ wallet, amount, asset }) =>
    ({ txId: await myWallet.transfer(asset, amount, wallet) }),
});

const charge = await noa.createPaymentLink({
  amount: 5000,
  currency: FiatCurrency.ARS,
  reference: "order-1",
});

charge.link; // URL de Checkout Pro
charge.qrImage; // PNG listo para incrustar
 
const noa = new NoaClient({
  etherfuse: { apiKey, environment: "sandbox" },
});

// El activo se resuelve en vivo, nunca se codifica
const catalog = await noa.etherfuse.lookup.stablebonds();

const quote = await noa.etherfuse.quotes.create({ /* ... */ });
const receipt = await quote.createOrder({
  bankAccountId,
  publicKey: "USER_SOLANA_ADDRESS",
});

receipt.createPixQr(); // QR listo para el usuario
 
const providers = await noa.koywe.getPaymentProviders(
  FiatCurrency.ARS,
); // WIREAR (CVU), QRI-AR (QR)...

const quote = await noa.koywe.getQuote({
  ramp: "onramp",
  fiatCurrency: FiatCurrency.ARS,
  amount: "10000",
  paymentMethodId: providers[0].id,
});

const order = await noa.koywe.createOnRampOrder({
  quoteId: quote.id,
  stellarAddress: "USER_STELLAR_ADDRESS",
});

order.interactiveUrl; // checkout / estado
 
const noa = new NoaClient({}); // sin proveedores

const toml = await noa.sep.fetchStellarToml(
  "testanchor.stellar.org",
);

// Las llaves privadas nunca salen de tu firmante
const jwt = await noa.sep.authenticateSep10({
  webAuthEndpoint: toml.WEB_AUTH_ENDPOINT,
  account,
  sign: (xdr, network) => myWallet.signTransaction(xdr, network),
});

const { url, id } = await noa.sep.startDeposit({
  transferServer: toml.TRANSFER_SERVER_SEP0024,
  jwt, assetCode: Asset.USDC, account,
});
