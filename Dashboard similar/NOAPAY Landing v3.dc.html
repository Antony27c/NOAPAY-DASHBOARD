quickstart.ts 
import { NoaPay, FiatCurrency } from "noa-pay";

const noa = new NoaPay({
  mercadopago: { accessToken, sandbox: true },
});

const payment = await noa.createPayment({
  amount: 5000,
  currency: FiatCurrency.ARS,
  description: "Order #1234",
});

payment.link;      // Checkout Pro
payment.qrImage;   // data URL, listo para <img>
payment.reference; // tuyo: lo que casan los webhooks
