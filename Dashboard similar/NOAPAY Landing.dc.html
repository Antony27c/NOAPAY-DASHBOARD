quickstart.ts 
import { NoaClient, FiatCurrency } from "noa-pay";

const noa = new NoaClient({
  mercadopago: { accessToken, sandbox: true },
  settlement: async ({ wallet, amount, asset }) =>
    ({ txId: await myWallet.transfer(asset, amount, wallet) }),
});

const charge = await noa.createPaymentLink({
  amount: 5000,
  currency: FiatCurrency.ARS,
  reference: "order-1",
});

charge.link; // URL de Checkout Pro
charge.qrImage; // PNG listo para incrustar
